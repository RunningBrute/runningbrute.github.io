```cpp
#include <iostream>

int main()
{
    auto emptyOne = [](){};
    emptyOne();
  
    int value = 42;
    auto captureDataByCopy = [value](){
    	std::cout << value << std::endl;
    };
    captureDataByCopy();
    
    auto captureDataByRef = [&value](){
    	std::cout << value << std::endl;
    };
    captureDataByRef();
  
    int param1 = 13;
    float param2 = 4.2;
    auto withParams = [](int a, float b){
    	std::cout << a << " and " << b << std::endl;
    };
    withParams(param1, param2);
    
}
```
